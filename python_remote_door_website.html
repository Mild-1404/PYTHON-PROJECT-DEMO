3. จำลองการทำงานของประตูไฟฟ้า

ทดลองกดปุ่มได้จริงในหน้าเว็บ: A เปิด, B ปิด, C หยุด ประตูจะเคลื่อนที่ทีละน้อยจนถึงปลายทาง

มุมมองประตูโรงรถ (ภาพจำลอง)
สถานะ: STOPตำแหน่ง: 0%

ประตูปิดสนิท พร้อมรับคำสั่ง

แผงควบคุมรีโมทจำลอง

กดปุ่มเพื่อเปลี่ยนสถานะประตู

ระบบเริ่มต้น: ประตูอยู่ที่ 0% และหยุดนิ่ง

หมายเหตุ: ปุ่ม “ขยับ 10%” ใช้ทดสอบการแสดงผลทีละขั้นเท่านั้น

4. โค้ด Python ที่ใช้จำลองประตู

ตัวอย่างนี้รันใน Terminal/Command Prompt ได้ ใช้ปุ่ม A, B, C เพื่อสั่งงาน และ Q เพื่อออกจากโปรแกรม การเคลื่อนที่จะจำลองเป็นช่วงเวลา

door_simulation.py
import time
import threading

# ตำแหน่งประตู: 0 = ปิดสนิท, 100 = เปิดสุด
door_position = 0
motor_direction = "STOP"
lock = threading.Lock()
running = True


def open_door():
    global motor_direction
    with lock:
        if door_position >= 100:
            motor_direction = "STOP"
            print("ประตูเปิดสุดแล้ว")
        else:
            motor_direction = "OPEN"
            print("กำลังเปิดประตู")


def close_door():
    global motor_direction
    with lock:
        if door_position <= 0:
            motor_direction = "STOP"
            print("ประตูปิดสนิทแล้ว")
        else:
            motor_direction = "CLOSE"
            print("กำลังปิดประตู")


def stop_door():
    global motor_direction
    with lock:
        motor_direction = "STOP"
    print("หยุดประตู")


def movement_loop():
    """จำลองการเคลื่อนที่ต่อเนื่องจนสุดทางหรือได้รับคำสั่งหยุด"""
    global door_position, motor_direction, running

    while running:
        with lock:
            if motor_direction == "OPEN":
                door_position = min(100, door_position + 5)
                print(f"ตำแหน่งประตู: {door_position}%")
                if door_position >= 100:
                    motor_direction = "STOP"
                    print("เปิดสุดแล้ว — หยุดอัตโนมัติ")

            elif motor_direction == "CLOSE":
                door_position = max(0, door_position - 5)
                print(f"ตำแหน่งประตู: {door_position}%")
                if door_position <= 0:
                    motor_direction = "STOP"
                    print("ปิดสนิทแล้ว — หยุดอัตโนมัติ")

        time.sleep(0.25)


# แยกการจำลองการเคลื่อนที่ไว้ในเธรดอีกตัว
worker = threading.Thread(target=movement_loop, daemon=True)
worker.start()

try:
    while True:
        print("\\nA = เปิด | B = ปิด | C = หยุด | Q = ออก")
        button = input("กดปุ่มจำลอง: ").strip().upper()

        if button == "A":
            open_door()
        elif button == "B":
            close_door()
        elif button == "C":
            stop_door()
        elif button == "Q":
            stop_door()
            break
        else:
            print("ไม่พบคำสั่ง กรุณากด A, B, C หรือ Q")

finally:
    running = False
    stop_door()
    print("ปิดโปรแกรมจำลอง")
หลักการของโค้ด: if/elif ตรวจสอบปุ่ม, ฟังก์ชัน open_door(), close_door() และ stop_door() จัดการคำสั่ง, ส่วน movement_loop() จำลองการเคลื่อนที่ทีละ 5% จนถึง 0% หรือ 100%

คำสั่งสำคัญในโค้ด

คำสั่งหน้าที่
defสร้างฟังก์ชันเพื่อแบ่งหน้าที่ของโปรแกรม
if / elifตรวจสอบว่าผู้ใช้กดปุ่มใด
whileทำให้ระบบจำลองทำงานต่อเนื่อง
min() / max()จำกัดตำแหน่งไม่ให้เกินช่วง 0–100%
threadingแยกการเคลื่อนที่จำลองออกจากการรอรับคำสั่ง
time.sleep()กำหนดช่วงเวลาระหว่างการอัปเดตตำแหน่ง

ข้อควรระวังก่อนพัฒนาเป็นของจริง

  • โค้ดนี้เป็นเพียงแบบจำลองบนคอมพิวเตอร์ ไม่ได้อ่านสัญญาณรีโมทจริงหรือสั่งมอเตอร์จริง
  • ก่อนใช้กับประตูจริง ต้องออกแบบวงจรควบคุมและระบบหยุดที่เหมาะสมกับมอเตอร์
  • ควรมี Limit Switch, เซ็นเซอร์ตรวจสิ่งกีดขวาง และปุ่มหยุดฉุกเฉินที่เชื่อถือได้
  • ห้ามต่อขา GPIO เข้ากับมอเตอร์หรือไฟบ้านโดยตรง การติดตั้งไฟฟ้าและชุดมอเตอร์ควรให้ช่างผู้ชำนาญตรวจสอบ
  • ระบบจริงต้องทดสอบกรณีไฟดับ สัญญาณขาด เซ็นเซอร์ขัดข้อง และคำสั่งเปิด/ปิดสวนทางกัน
